viewof selected_country = Inputs.select(
  countries,
  {
    label: "Choisir un pays :",
    value: "France"
  }
)

viewof unite = Inputs.radio(types_unit, {
  label: "Unité à afficher en y :",
  format: x => x.label,
  value: types_unit[1]   
})
country_health_spending = health_spending.filter(
  d => d.country_name === selected_country &&
    d.unit === unite.name
  )
  .map(
    row => ({
      ...row,
      value: row.unit === "constant 2023 US$" ? row.value / 1_000_000_000 : row.value
    })
  )
values = country_health_spending.map(row => row.value)
years = country_health_spending.map(row => row.year.getFullYear())

maxValue = Math.max(...values)
minValue = Math.min(...values)
